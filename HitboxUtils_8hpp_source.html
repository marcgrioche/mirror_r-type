23inline bool aabb_overlap_world(const Position& pa, const Hitbox& ha,
24 const Position& pb, const Hitbox& hb)
25{
26 float axmin = pa.v.x + ha.offset_x;
27 float aymin = pa.v.y + ha.offset_y;
28 float axmax = axmin + ha.width;
29 float aymax = aymin + ha.height;
30
31 float bxmin = pb.v.x + hb.offset_x;
32 float bymin = pb.v.y + hb.offset_y;
33 float bxmax = bxmin + hb.width;
34 float bymax = bymin + hb.height;
35
36 // no overlap if one is fully left/right/top/bottom of the other
37 // treat edge-touching as non-collision (use <=). Change to < to include touching.
38 if (axmax <= bxmin)
39 return false;
40 if (bxmax <= axmin)
41 return false;
42 if (aymax <= bymin)
43 return false;
44 if (bymax <= aymin)
45 return false;
46 return true;
47}

57inline bool entities_collide(Registry& registry, Entity a, Entity b)
58{
59 if (!registry.has<Position>(a) || !registry.has<Hitbox>(a))
60 return false;
61 if (!registry.has<Position>(b) || !registry.has<Hitbox>(b))
62 return false;
63
64 const Position& pa = registry.get<Position>(a);
65 const Hitbox& ha = registry.get<Hitbox>(a);
66 const Position& pb = registry.get<Position>(b);
67 const Hitbox& hb = registry.get<Hitbox>(b);
68
69 return aabb_overlap_world(pa, ha, pb, hb);
